    let saldo = monto;
    const tabla = [];
    const fechaInicio = new Date();

    for (let k = 1; k <= plazoMeses; k++) {
      const interes = saldo * i_mensual;
      const capital = cuota - interes;
      saldo = Math.max(0, saldo - capital);

      const fechaPago = new Date(fechaInicio);
      fechaPago.setMonth(fechaPago.getMonth() + k);

        num_cuota: k,
        fecha_vencimiento: fechaPago.toISOString().split('T')[0],

  return (
    <div className="space-y-6">
      
      {/* TOAST SYSTEM */}
      {notificacion && (
        <div className="fixed top-4 right-4 left-4 sm:left-auto sm:max-w-md z-50 animate-fade-in">
          <div className={`rounded-xl p-4 shadow-2xl border bg-slate-900 text-white flex items-start gap-3 ${
            notificacion.tipo === 'success' ? 'border-emerald-500' : notificacion.tipo === 'warning' ? 'border-amber-500' : 'border-rose-500'
          }`}>
            <div className="mt-0.5 shrink-0">
              {notificacion.tipo === 'success' ? <CheckCircle2 className="w-5 h-5 text-emerald-400" /> : <ShieldAlert className="w-5 h-5 text-rose-400" />}
            </div>
            <div className="flex-1 min-w-0">
              <h4 className="text-sm font-semibold truncate">{notificacion.titulo}</h4>
              <p className="text-xs text-slate-300 mt-0.5 leading-relaxed">{notificacion.mensaje}</p>
            </div>
            <button onClick={() => setNotificacion(null)} className="text-slate-400 hover:text-white">&times;</button>
          </div>
        </div>
      )}

      {/* HEADER PRINCIPAL Y NAVEGACIÓN ENTRE VISTAS */}
      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4 bg-white p-5 rounded-xl border border-slate-200 shadow-sm">
        <div>
          <div className="flex items-center gap-2">
            <ShieldCheck className="w-6 h-6 text-slate-800" />
            <h1 className="text-xl font-bold text-slate-800">Módulo de Aprobaciones (CU07 / CU08)</h1>
          </div>
          <p className="text-xs text-slate-500 mt-1">Gestión del Jefe de Créditos · Atribuciones (RN04), Contratos (RN05), Cronogramas (RN06) y Estadísticas.</p>
        </div>

        {/* NAVEGACIÓN ENTRE VISTAS DE PESTAÑAS */}
        <div className="flex bg-slate-100 p-1 rounded-lg border border-slate-200 self-start sm:self-auto overflow-x-auto max-w-full">
          <button
            onClick={() => setVistaActiva('PENDIENTES')}
            className={`px-4 py-2 text-xs font-bold rounded-md transition-colors whitespace-nowrap ${
              vistaActiva === 'PENDIENTES' ? 'bg-slate-900 text-white shadow-sm' : 'text-slate-600 hover:text-slate-900'
            }`}
          >
            Bandeja({solicitudesPendientes.length})
          </button>
          <button
            onClick={() => setVistaActiva('HISTORIAL')}
            className={`px-4 py-2 text-xs font-bold rounded-md transition-colors whitespace-nowrap ${
              vistaActiva === 'HISTORIAL' ? 'bg-slate-900 text-white shadow-sm' : 'text-slate-600 hover:text-slate-900'
            }`}
          >
            Historial ({historicoResoluciones.length})
          </button>
          <button
            onClick={() => setVistaActiva('ESTADISTICAS')}
            className={`px-4 py-2 text-xs font-bold rounded-md transition-colors whitespace-nowrap ${
              vistaActiva === 'ESTADISTICAS' ? 'bg-slate-900 text-white shadow-sm' : 'text-slate-600 hover:text-slate-900'
            }`}
          >
            Estadística
          </button>
        </div>
      </div>

      {/* =========================================================================
          VISTA 1: BANDEJA DE DECISIONES DE EVALUACIÓN (CU07 / CU08)
         ========================================================================= */}
      {vistaActiva === 'PENDIENTES' && (
        <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
          
          {/* PANEL IZQUIERDO: LISTA DE SOLICITUDES PENDIENTES */}
          <div className="lg:col-span-1 bg-white rounded-xl border border-slate-200 p-5 shadow-sm space-y-4">
            <div className="flex justify-between items-center">
              <h3 className="text-xs font-bold text-slate-400 uppercase tracking-wider">Pendientes de Decisión</h3>
              <span className="text-[10px] font-bold bg-sky-100 text-sky-800 px-2 py-0.5 rounded-full">
                {solicitudesPendientes.length} por revisar
              </span>
            </div>
            
            <div className="space-y-3">
              {solicitudesPendientes.length > 0 ? (
                solicitudesPendientes.map((sol) => (
                  <div 
                    key={sol.id_solicitud}
                    onClick={() => setSolicitudSel(sol)}
                    className={`p-4 rounded-xl border transition-all cursor-pointer ${
                      solicitudSel?.id_solicitud === sol.id_solicitud 
                        ? 'border-slate-800 bg-slate-50 shadow-sm ring-1 ring-slate-800' 
                        : 'border-slate-200 hover:border-slate-300 bg-white'
                    }`}
                  >
                    <div className="flex justify-between items-start mb-2">
                      <span className="text-[10px] font-mono bg-sky-100 text-sky-800 px-2 py-0.5 rounded font-bold">
                        SOL-#{sol.id_solicitud}
                      </span>
                      <span className="text-xs font-bold text-slate-900">Bs. {sol.monto_solicitado.toLocaleString()}</span>
                    </div>
                    <h4 className="font-semibold text-slate-900 text-sm">{sol.cliente}</h4>
                    <p className="text-xs text-slate-500 mt-0.5">C.I.: {sol.ci} | {sol.destino_credito}</p>

                    <div className="mt-3 pt-3 border-t border-slate-100 flex justify-between items-center text-xs">
                      <span className="font-bold text-emerald-700 bg-emerald-50 px-2 py-0.5 rounded">
                        {sol.evaluacion.resultado}
                      </span>
                      <span className="text-slate-800 font-bold flex items-center gap-1">
                        Revisar <ArrowRight className="w-3.5 h-3.5" />
                      </span>
                    </div>
                  </div>
                ))
              ) : (
                <div className="p-8 text-center text-slate-400 text-xs sm:text-sm">
                  No existen solicitudes pendientes en la bandeja de evaluación.
                </div>
              )}
            </div>
          </div>

          {/* PANEL DERECHO: REVISIÓN INTEGRAL 360° Y ACCIÓN OFICIAL */}
          <div className="lg:col-span-2">
            {solicitudSel ? (
              <div className="bg-white rounded-xl border border-slate-200 shadow-sm overflow-hidden p-6 space-y-6">
                
                {/* CABECERA RESUMEN SOLICITUD */}
                <div className="flex justify-between items-start border-b border-slate-200 pb-4">
                  <div>
                    <span className="text-xs font-mono text-slate-400">Expediente SOL-#{solicitudSel.id_solicitud}</span>
                    <h2 className="text-lg font-bold text-slate-900">{solicitudSel.cliente}</h2>
                    <p className="text-xs text-slate-500">C.I.: {solicitudSel.ci} | Solicitado: {solicitudSel.fecha_solicitud}</p>
                  </div>

                  <div className="text-right">
                    <span className="text-xs text-slate-400 block font-medium">Monto Aprobación</span>
                    <span className="text-2xl font-black text-slate-900">Bs. {solicitudSel.monto_solicitado.toLocaleString()}</span>
                    <span className="text-xs text-slate-500 block">{solicitudSel.plazo_meses} meses @ {solicitudSel.tasa_interes_anual}%</span>
                  </div>
                </div>

                {/* ADVERTENCIA DE VERIFICACIÓN DE ATRIBUCIÓN (RN04) */}
                {solicitudSel.monto_solicitado > LIMITE_ATRIBUCION_JEFE && (
                  <div className="flex items-start gap-3 bg-amber-50 border border-amber-200 p-3.5 rounded-xl text-xs text-amber-900">
                    <AlertTriangle className="w-5 h-5 text-amber-600 shrink-0 mt-0.5" />
                    <div>
                      <strong className="block font-bold">RN04 - Exceso de Atribución Crediticia:</strong>
                      El monto (Bs. {solicitudSel.monto_solicitado.toLocaleString()}) supera su atribución máxima (Bs. {LIMITE_ATRIBUCION_JEFE.toLocaleString()}). Al intentar aprobar se requerirá confirmación de escalamiento.
                    </div>
                  </div>
                )}

                {/* EVALUACIÓN DE CAPACIDAD Y RIESGO (CU04 / CU05) */}
                <div>
                  <h3 className="text-xs font-bold text-slate-400 uppercase tracking-wider mb-3">Dictamen del Analista de Crédito</h3>
                  <div className="grid grid-cols-1 sm:grid-cols-4 gap-3 mb-3 text-xs">
                    <div className="bg-slate-50 p-3 rounded-lg border border-slate-200">
                      <span className="text-slate-500 block">Ingresos Verificados</span>
                      <span className="font-bold text-slate-900">Bs. {solicitudSel.evaluacion.ingresos.toLocaleString()}</span>
                    </div>
                    <div className="bg-slate-50 p-3 rounded-lg border border-slate-200">
                      <span className="text-slate-500 block">Gastos Registrados</span>
                      <span className="font-bold text-slate-900">Bs. {solicitudSel.evaluacion.gastos.toLocaleString()}</span>
                    </div>
                    <div className="bg-slate-50 p-3 rounded-lg border border-slate-200">
                      <span className="text-slate-500 block">Capacidad de Pago</span>
                      <span className="font-bold text-slate-900">Bs. {solicitudSel.evaluacion.capacidad_pago.toLocaleString()}</span>
                    </div>
                    <div className="bg-emerald-50 p-3 rounded-lg border border-emerald-200 text-emerald-900">
                      <span className="block text-[11px] font-semibold">Cobertura (DSCR)</span>
                      <span className="text-lg font-black">{solicitudSel.evaluacion.cobertura_dscr}</span>
                    </div>
                  </div>

                  <div className="bg-slate-50 p-4 rounded-xl border border-slate-200 text-xs space-y-1.5">
                    <div className="flex justify-between font-semibold text-slate-700">
                      <span>Analista Evaluador: {solicitudSel.evaluacion.analista}</span>
                      <span>Nivel Riesgo: <strong className="text-slate-900">{solicitudSel.evaluacion.nivel_riesgo}</strong></span>
                    </div>
                    <p className="text-slate-600 leading-relaxed italic">"{solicitudSel.evaluacion.observaciones_analista}"</p>
                  </div>
                </div>

                {/* HISTORIAL BURÓ (CU03) Y GARANTÍA (CU06) */}
                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 text-xs">
                  <div className="border border-slate-200 p-4 rounded-xl bg-slate-50">
                    <h4 className="font-bold text-slate-700 mb-2">Central de Riesgos (ASFI)</h4>
                    <p><strong>Calificación:</strong> <span className="text-emerald-700 font-bold">{solicitudSel.evaluacion.calificacion_asfi}</span></p>
                    <p><strong>Score Buró:</strong> {solicitudSel.evaluacion.score_buro} / 850</p>
                    <p><strong>Ref. Consulta:</strong> {solicitudSel.evaluacion.ref_asfi}</p>
                  </div>

                  <div className="border border-slate-200 p-4 rounded-xl bg-slate-50">
                    <h4 className="font-bold text-slate-700 mb-2">Garantía Colateral (CU06)</h4>
                    <p><strong>Tipo:</strong> {solicitudSel.garantia_detalles.tipo}</p>
                    <p><strong>Valor Comercial:</strong> Bs. {solicitudSel.garantia_detalles.valor_comercial.toLocaleString()}</p>
                    <p><strong>Cobertura:</strong> <span className="text-emerald-700 font-bold">{solicitudSel.garantia_detalles.cobertura_garantia}</span></p>
                  </div>
                </div>

                {/* BOTONES DE ACCIÓN (CU07 / CU08) */}
                <div className="flex flex-col sm:flex-row justify-end gap-3 pt-4 border-t border-slate-200">
                  <button
                    onClick={() => setModalAccion('CONFIRMAR_RECHAZAR')}
                    className="inline-flex items-center justify-center gap-2 px-5 py-2.5 bg-rose-50 hover:bg-rose-100 text-rose-700 border border-rose-200 rounded-lg text-xs font-bold transition-colors"
                  >
                    <XCircle className="w-4 h-4" />
                    Rechazar Crédito (CU08)
                  </button>

                  <button
                    onClick={() => setModalAccion('CONFIRMAR_APROBAR')}
                    className="inline-flex items-center justify-center gap-2 px-6 py-2.5 bg-slate-900 hover:bg-slate-800 text-white rounded-lg text-xs font-bold shadow transition-colors"
                  >
                    <CheckCircle2 className="w-4 h-4 text-emerald-400" />
                    Aprobar Crédito y Generar Documentos (CU07)
                  </button>
                </div>

              </div>
            ) : (
              // PANEL DERECHO EN BLANCO SOLICITADO
              <div className="bg-white rounded-xl border border-slate-200 p-12 text-center text-slate-400 text-xs sm:text-sm shadow-sm flex flex-col items-center justify-center min-h-[400px]">
                <FileText className="w-12 h-12 text-slate-300 mb-3" />
                <p className="font-semibold text-slate-600">Seleccione una solicitud de la lista izquierda para revisar el expediente integral.</p>
                <p className="text-xs text-slate-400 mt-1">El expediente incluye evaluación de capacidad, riesgo ASFI y garantías colaterales.</p>
              </div>
            )}
          </div>

        </div>
      )}

      {/* =========================================================================
          VISTA 2: HISTORIAL DE RESOLUCIONES (REPORTE Y SEGUIMIENTO)
         ========================================================================= */}
      {vistaActiva === 'HISTORIAL' && (
        <div className="space-y-4">
          <div className="bg-white p-4 rounded-xl border border-slate-200 shadow-sm flex flex-col sm:flex-row gap-3 items-center justify-between">
            <div className="flex items-center gap-2 w-full sm:w-auto">
              <Search className="w-4 h-4 text-slate-400 shrink-0" />
              <input
                type="text"
                placeholder="Buscar en historial por Cliente, C.I. o N° Solicitud..."
                value={busquedaHistorial}
                onChange={(e) => setBusquedaHistorial(e.target.value)}
                className="w-full sm:w-80 text-xs sm:text-sm border-none focus:outline-none text-slate-800 placeholder-slate-400"
              />
            </div>

            <div className="flex overflow-x-auto w-full sm:w-auto gap-1 text-xs">
              {['TODOS', 'APROBADA', 'RECHAZADA'].map((est) => (
                <button
                  key={est}
                  onClick={() => setFiltroEstadoHistorial(est)}
                  className={`px-3 py-1.5 rounded-lg font-semibold transition-colors ${
                    filtroEstadoHistorial === est 
                      ? 'bg-slate-800 text-white' 
                      : 'bg-slate-50 text-slate-600 border border-slate-200 hover:bg-slate-100'
                  }`}
                >
                  {est}
                </button>
              ))}
            </div>
          </div>

          <div className="bg-white rounded-xl border border-slate-200 shadow-sm overflow-hidden">
            <div className="overflow-x-auto">
              <table className="w-full text-left text-xs sm:text-sm">
                <thead className="bg-slate-50 border-b border-slate-200 text-slate-600 font-semibold uppercase text-[11px]">
                  <tr>
                    <th className="p-4">Solicitud / Cliente</th>
                    <th className="p-4">Monto & Términos</th>
                    <th className="p-4">Estado Final</th>
                    <th className="p-4">Resolución / Documentos</th>
                    <th className="p-4">Aprobado Por</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-slate-100 text-slate-700">
                  {historicoFiltrado.length > 0 ? (
                    historicoFiltrado.map((res) => (
                      <tr key={res.id_solicitud} className="hover:bg-slate-50">
                        <td className="p-4">
                          <span className="font-mono text-[10px] bg-slate-100 text-slate-700 px-2 py-0.5 rounded font-bold block w-max mb-1">
                            SOL-#{res.id_solicitud}
                          </span>
                          <div className="font-bold text-slate-900">{res.cliente}</div>
                          <div className="text-xs text-slate-400">{res.ci}</div>
                        </td>
                        <td className="p-4">
                          <div className="font-bold text-slate-900">Bs. {res.monto.toLocaleString()}</div>
                          <div className="text-xs text-slate-500">{res.plazo_meses} meses @ {res.tasa}%</div>
                        </td>
                        <td className="p-4">
                          <span className={`inline-flex items-center px-2.5 py-1 rounded-full text-xs font-bold ${
                            res.estado === 'APROBADA' ? 'bg-emerald-50 text-emerald-800 border border-emerald-200' : 'bg-rose-50 text-rose-800 border border-rose-200'
                          }`}>
                            {res.estado}
                          </span>
                        </td>
                        <td className="p-4">
                          {res.estado === 'APROBADA' ? (
                            <div>
                              <span className="font-mono text-xs text-emerald-700 font-bold block">{res.num_contrato}</span>
                              <span className="text-[11px] text-slate-400">Contrato y Cronograma OK</span>
                            </div>
                          ) : (
                            <span className="text-xs text-rose-600 italic block">{res.motivo_rechazo}</span>
                          )}
                        </td>
                        <td className="p-4 text-xs text-slate-600">
                          {res.aprobado_por}
                          <div className="text-[10px] text-slate-400">{res.fecha_resolucion}</div>
                        </td>
                      </tr>
                    ))
                  ) : (
                    <tr>
                      <td colSpan="5" className="p-8 text-center text-slate-400 text-sm">
                        No hay resoluciones registradas en el historial.
                      </td>
                    </tr>
                  )}
                </tbody>
              </table>
            </div>
          </div>
        </div>
      )}

      {/* =========================================================================
          VISTA 3: DASHBOARD ESTADÍSTICO Y REPORTES POR MES/DÍA/AÑO
         ========================================================================= */}
      {vistaActiva === 'ESTADISTICAS' && (
        <div className="space-y-6">
          
          {/* FILTRO TEMPORAL */}
          <div className="flex justify-between items-center bg-white p-4 rounded-xl border border-slate-200 shadow-sm">
            <span className="text-xs font-bold text-slate-700 uppercase tracking-wider">Período Estadístico</span>
            <div className="flex bg-slate-100 p-1 rounded-lg border border-slate-200 text-xs">
              <button onClick={() => setFiltroEstadistica('DIA')} className={`px-3 py-1 rounded font-bold ${filtroEstadistica === 'DIA' ? 'bg-slate-900 text-white' : 'text-slate-600'}`}>Hoy / Día</button>
              <button onClick={() => setFiltroEstadistica('MES')} className={`px-3 py-1 rounded font-bold ${filtroEstadistica === 'MES' ? 'bg-slate-900 text-white' : 'text-slate-600'}`}>Este Mes (Marzo 2026)</button>
              <button onClick={() => setFiltroEstadistica('ANIO')} className={`px-3 py-1 rounded font-bold ${filtroEstadistica === 'ANIO' ? 'bg-slate-900 text-white' : 'text-slate-600'}`}>Año 2026</button>
            </div>
          </div>

          {/* TARJETAS DE KPIs */}
          <div className="grid grid-cols-1 sm:grid-cols-4 gap-4">
            <div className="bg-white p-5 rounded-xl border border-slate-200 shadow-sm">
              <span className="text-xs text-slate-500 font-medium block">Total Solicitudes Dictaminadas</span>
              <span className="text-3xl font-black text-slate-900 mt-1 block">{stats.totalProcesados}</span>
              <span className="text-[11px] text-slate-400 mt-1 block">Bandeja de decisiones</span>
            </div>

            <div className="bg-white p-5 rounded-xl border border-slate-200 shadow-sm">
              <span className="text-xs text-slate-500 font-medium block">Tasa de Aprobación</span>
              <span className="text-3xl font-black text-emerald-600 mt-1 block">{stats.tasaAprobacion}%</span>
              <span className="text-[11px] text-slate-400 mt-1 block">{stats.totalAprobados} aprobadas / {stats.totalRechazados} rechazadas</span>
            </div>

            <div className="bg-white p-5 rounded-xl border border-slate-200 shadow-sm">
              <span className="text-xs text-slate-500 font-medium block">Volumen Colocado Aprobado</span>
              <span className="text-2xl font-black text-slate-900 mt-1 block">Bs. {stats.montoTotalAprobado.toLocaleString()}</span>
              <span className="text-[11px] text-emerald-600 mt-1 block">Cartera aprobada</span>
            </div>

            <div className="bg-white p-5 rounded-xl border border-slate-200 shadow-sm">
              <span className="text-xs text-slate-500 font-medium block">Volumen Rechazado</span>
              <span className="text-2xl font-black text-rose-600 mt-1 block">Bs. {stats.montoTotalRechazado.toLocaleString()}</span>
              <span className="text-[11px] text-rose-400 mt-1 block">Riesgo no admitido</span>
            </div>
          </div>

          {/* ANÁLISIS BARRAS Y DISTRIBUCIÓN */}
          <div className="bg-white p-6 rounded-xl border border-slate-200 shadow-sm space-y-4">
            <h3 className="text-sm font-bold text-slate-800">Distribución de Decisiones de Crédito</h3>
            
            <div className="space-y-3 text-xs">
              <div>
                <div className="flex justify-between font-semibold mb-1">
                  <span className="text-emerald-700">Solicitudes Aprobadas ({stats.totalAprobados})</span>
                  <span>Bs. {stats.montoTotalAprobado.toLocaleString()}</span>
                </div>
                <div className="w-full bg-slate-100 rounded-full h-3 overflow-hidden">
                  <div className="bg-emerald-500 h-3 rounded-full" style={{ width: `${stats.tasaAprobacion}%` }}></div>
                </div>
              </div>

              <div>
                <div className="flex justify-between font-semibold mb-1">
                  <span className="text-rose-700">Solicitudes Rechazadas ({stats.totalRechazados})</span>
                  <span>Bs. {stats.montoTotalRechazado.toLocaleString()}</span>
                </div>
                <div className="w-full bg-slate-100 rounded-full h-3 overflow-hidden">
                  <div className="bg-rose-500 h-3 rounded-full" style={{ width: `${100 - parseFloat(stats.tasaAprobacion)}%` }}></div>
                </div>
              </div>
            </div>
          </div>

        </div>
      )}

      {/* =========================================================================
          MODAL 1: CONFIRMAR APROBACIÓN OFICIAL (CU07)
         ========================================================================= */}
      {modalAccion === 'CONFIRMAR_APROBAR' && solicitudSel && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-900/60 backdrop-blur-sm">
          <div className="bg-white rounded-xl shadow-2xl w-full max-w-md overflow-hidden">
            <div className="p-4 bg-slate-900 text-white flex justify-between items-center">
              <h3 className="font-bold text-sm">Confirmar Aprobación Oficial (CU07)</h3>
              <button onClick={() => setModalAccion(null)} className="text-slate-400 hover:text-white font-bold">&times;</button>
            </div>

            <div className="p-6 space-y-4 text-xs sm:text-sm">
              <p className="text-slate-700">
                ¿Está seguro de aprobar el crédito de <strong className="text-slate-900">Bs. {solicitudSel.monto_solicitado.toLocaleString()}</strong> para <strong className="text-slate-900">{solicitudSel.cliente}</strong>?
              </p>

              <div className="bg-slate-50 border border-slate-200 p-3.5 rounded-lg space-y-1 text-xs text-slate-600">
                <p>&bull; Estado cambiará a <strong>APROBADA</strong> en Supabase.</p>
                <p>&bull; Se generará el Contrato Oficial numerado (RN05).</p>
                <p>&bull; Se generará la Tabla de Amortización Francesa (RN06).</p>
              </div>

              <div className="flex justify-end gap-2 pt-2 border-t border-slate-200">
                <button type="button" onClick={() => setModalAccion(null)} className="px-4 py-2 text-xs font-semibold text-slate-600 bg-white border border-slate-300 rounded-lg">Cancelar</button>
                <button 
                  type="button" 
                  onClick={handleEjecutarAprobacionOficial}
                  className="px-5 py-2 bg-slate-900 hover:bg-slate-800 text-white font-bold rounded-lg shadow text-xs"
                >
                  Confirmar Aprobación
                </button>
              </div>
            </div>
          </div>
        </div>
      )}

      {/* =========================================================================
          MODAL 2: CONFIRMAR RECHAZO DE CRÉDITO (CU08)
         ========================================================================= */}
      {modalAccion === 'CONFIRMAR_RECHAZAR' && solicitudSel && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-900/60 backdrop-blur-sm">

1. Casos de uso mejorados para tu documentación
CU09 – Generar documentación contractual del crédito
CU09 – Generar documentación contractual

Formalización documental de un crédito aprobado

Actor principal: Sistema de gestión de créditos.

Actores secundarios: Oficial o responsable de créditos y cliente beneficiario.

Objetivo: Generar, registrar y poner a disposición del personal autorizado la documentación correspondiente al crédito aprobado, garantizando la consistencia de los datos y la trazabilidad del proceso.

Precondiciones

La solicitud de crédito se encuentra en estado APROBADA.

Existe un registro de crédito asociado en la tabla credito.

El crédito dispone de un número único, monto aprobado, tasa de interés y plazo válidos.

Se dispone de los datos de identificación del cliente y de la información institucional necesaria para elaborar los documentos.

Las garantías y sus documentos de respaldo se encuentran registrados cuando correspondan.

Flujo principal

El sistema recupera la información de la solicitud aprobada, el cliente, el crédito, las garantías y la configuración institucional.

Verifica que los datos obligatorios sean consistentes y que el crédito no tenga documentación contractual previamente generada.

Genera un número único de contrato y registra la fecha de emisión.

Elabora el Acuerdo de Condiciones de Crédito, incluyendo los datos del cliente, monto aprobado, tasa de interés, plazo, cuota estimada o definitiva, fechas de vencimiento, garantías y condiciones particulares.

Genera la documentación complementaria correspondiente, incluido el contrato privado de préstamo, conforme a las plantillas institucionales y las cláusulas autorizadas.

Incorpora el membrete institucional, la razón social, el NIT, los datos de contacto y la identificación del documento.

Presenta los documentos para revisión, impresión o descarga en PDF por parte del personal autorizado.

Registra el contrato en la tabla contrato, asociado al crédito mediante id_credito, con su número, fecha de firma cuando corresponda y estado documental.

Confirma la generación y disponibilidad de la documentación.

Flujos alternativos y excepciones

A1. Datos incompletos: el sistema informa los campos faltantes y no genera documentos definitivos.

A2. Crédito no aprobado o inexistente: se cancela la operación y se informa que el crédito no cumple las condiciones para formalizarse.

A3. Contrato duplicado: el sistema identifica el contrato existente y evita generar otro para el mismo crédito.

A4. Garantías incompletas: cuando sean obligatorias, se suspende la formalización hasta completar su información y respaldo documental.

A5. Error al generar o guardar documentos: se informa el error y se evita dejar registros parciales o inconsistentes.

A6. Firma pendiente: el contrato se mantiene en estado pendiente de firma; no se registra como firmado hasta que exista constancia válida.

Postcondiciones

Existe un registro único del contrato asociado al crédito.

La documentación contractual está generada y disponible para revisión o impresión.

El estado documental refleja la situación real del proceso.

Se conserva la trazabilidad de la emisión y, cuando corresponda, de la firma.

Importante: tu tabla contrato solo permite almacenar el número, el crédito, la fecha de firma y el estado. Para registrar también la fecha de emisión, el archivo PDF, la fecha de desembolso documental y la trazabilidad de las versiones, sería necesario ampliar el modelo de datos.

CU10 – Generar cronograma de amortización del crédito
CU10 – Generar cronograma de amortización

Determinación de cuotas y fechas de pago

Actor principal: Sistema de gestión de créditos.

Actores secundarios: Oficial de créditos y cliente beneficiario.

Objetivo: Calcular y registrar el plan de pagos del crédito, detallando la distribución de capital e intereses, las fechas de vencimiento y el saldo de capital después de cada cuota.

Precondiciones

Existe un crédito aprobado y registrado en la tabla credito.

El monto aprobado, la tasa de interés y el plazo son válidos.

Se ha definido la fecha inicial de cálculo y la periodicidad de pago.

Se ha establecido el sistema de amortización y las reglas de redondeo aplicables.

Los parámetros necesarios para determinar la primera fecha de vencimiento están disponibles.

Flujo principal

El sistema recupera el número de crédito, el monto aprobado, la tasa de interés, el plazo y la información necesaria para establecer las fechas de pago.

Valida los parámetros financieros y la ausencia de un cronograma duplicado para la misma versión del crédito.

Determina la tasa periódica y el método de amortización configurado.

Calcula el importe de la cuota y desglosa cada pago en capital e intereses.

Determina la fecha de vencimiento de cada cuota, de acuerdo con la periodicidad y las reglas de calendario establecidas.

Calcula el saldo de capital después de cada pago, aplicando las reglas de precisión monetaria.

Registra las cuotas en la tabla cuota, incluyendo número de cuota, vencimiento, capital, interés, importe total, saldo y estado inicial.

Verifica que el cronograma esté completo, que las cuotas correspondan al plazo y que los importes y saldos sean consistentes.

Genera la tabla de amortización en formato institucional membretado, vinculada al número de crédito y al contrato correspondiente.

Presenta el documento para su consulta, impresión o descarga en PDF.

Flujos alternativos y excepciones

A1. Parámetros inválidos: se informa el campo incorrecto y se cancela el cálculo.

A2. Crédito inexistente o no aprobado: no se genera el cronograma.

A3. Fecha de vencimiento no válida: se aplica la regla de calendario definida o se solicita corregir la fecha inicial.

A4. Cronograma existente: se impide la duplicación de cuotas; cualquier modificación requiere un procedimiento autorizado de actualización o refinanciamiento.

A5. Diferencias por redondeo: se ajusta la última cuota conforme a las reglas financieras aprobadas, sin alterar arbitrariamente los importes.

A6. Error de persistencia: se revierte la operación para evitar que queden cuotas incompletas.

Postcondiciones

Las cuotas quedan registradas y relacionadas con el crédito mediante id_credito.

El cronograma refleja las condiciones financieras aprobadas.

Los saldos y los importes de capital e intereses son consistentes con el método de amortización utilizado.

La tabla de amortización está disponible para consulta, impresión o descarga.

3. Estructura profesional de los tres documentos

Te recomiendo que el sistema genere un expediente documental de tres hojas o secciones principales, con identidad visual institucional uniforme, numeración de páginas y datos consistentes en todos los documentos.

Documento 1. Acuerdo de Condiciones de Crédito / Comprobante de desembolso

Hoja 1 · Resumen de condiciones

Debe presentar de manera clara las condiciones financieras del crédito y dejar constancia del desembolso cuando este se haya realizado.

Membrete: logotipo, razón social, NIT, dirección y teléfono.

Identificación: número de contrato, número de crédito, número de solicitud y fecha de emisión.

Datos del cliente: nombre completo, CI y domicilio, si corresponde.

Condiciones financieras: monto aprobado, monto desembolsado, tasa anual, plazo, cuota, moneda y sistema de amortización.

Plan de pagos: fecha de la primera cuota, periodicidad y día habitual de vencimiento mensual.

Garantías: tipo, descripción, valor de avalúo y documentos de respaldo.

Condiciones particulares: obligaciones de pago, intereses, penalidades aplicables y referencias al contrato.

Firmas: nombre, CI, firma del cliente y firma del representante autorizado; fecha y lugar.

Documento 2. Tabla de amortización

Hoja 2 y páginas adicionales si son necesarias

Presenta el detalle de las cuotas para que el cliente y la entidad puedan verificar los importes y vencimientos.

Membrete institucional e identificación del titular.

Número de crédito, monto desembolsado, tasa de interés, plazo y fecha de inicio.

Método de amortización y periodicidad de pago.

Tabla con número de cuota, fecha de vencimiento, capital, interés, cuota total y saldo pendiente.

Resumen de cuotas, total de intereses y total a pagar según el método de cálculo.

Observaciones sobre fechas de vencimiento, pagos anticipados y posibles diferencias por redondeo.

Referencia al contrato y espacio de validación o firma cuando la política institucional lo requiera.

Documento 3. Contrato privado de préstamo de dinero

Documento jurídico · Extensión según las cláusulas

Formaliza los compromisos y obligaciones de las partes, sujeto a revisión jurídica y a la normativa aplicable.

Identificación completa de la entidad prestamista y del prestatario.

Antecedentes y objeto del contrato.

Monto del préstamo, moneda, condiciones de entrega y destino.

Tasa de interés, plazo, forma de pago y remisión al cronograma.

Obligaciones de las partes y condiciones de incumplimiento.

Garantías constituidas, cuando corresponda, y referencias a sus documentos.

Pagos anticipados, modificaciones, tratamiento de mora y gastos permitidos.

Domicilios para notificaciones, solución de controversias y cláusulas aplicables.

Lugar, fecha, firmas, nombres y documentos de identidad de los firmantes; formalidades adicionales cuando sean exigibles.

Las imágenes son referencias visuales del estilo documental, no formularios oficiales de BancoSol. Puedes utilizarlas como orientación para diseñar una plantilla propia de tu institución.

Reglas que conviene aplicar a los documentos

No confundir aprobación con desembolso. Un crédito aprobado no significa necesariamente que el dinero ya fue entregado. El comprobante de desembolso debe generarse o completarse cuando exista constancia de la operación.

No confundir emisión con firma. Un contrato preparado para firma debe distinguirse de uno formalizado.

Usar el monto aprobado, no el solicitado. Los documentos deben tomar las condiciones definitivas de credito.

No calcular la cuota únicamente en la interfaz. El cronograma debe calcularse y guardarse con reglas financieras consistentes.

No inventar información. Si faltan datos de identidad, garantía o desembolso, se deben señalar antes de emitir el documento definitivo.

4. Ajustes necesarios en tu código React

En el código que compartiste hay una diferencia relevante entre la demostración y una implementación real conectada a Supabase.

Actualmente, handleEjecutarAprobacionOficial:

Trabaja con datos simulados en memoria.

Genera el número de contrato mediante un valor aleatorio.

Calcula el cronograma en el navegador.

Usa el monto solicitado como si fuera el monto aprobado.

No inserta registros reales en las tablas credito, contrato y cuota.

No comprueba si ya existe un contrato o un cronograma para ese crédito.

No genera tres documentos PDF independientes ni registra una firma efectiva.

Por tanto, aunque la interfaz muestre que los documentos se generaron, eso no acredita que hayan quedado registrados en la base de datos.

Flujo recomendado para la implementación

1. Aprobación del crédito

Validar decisión y atribuciones del aprobador

2. Registrar o verificar credito

Monto, tasa, plazo, número y estado aprobados

3. Generar contrato y cuota

Transacción atómica y validación de duplicados

4. Elaborar los tres documentos

Condiciones, amortización y contrato privado

5. Revisión, firma y desembolso

Registrar cada evento cuando efectivamente ocurra

Para mantener la integridad, la operación de creación del crédito, contrato y cuotas debería gestionarse mediante una función transaccional en PostgreSQL o un servicio de backend confiable. La generación de PDF puede realizarse después de confirmar los registros.

6. Recomendación final para el proyecto

Yo mantendría CU09 y CU10 como dos casos de uso independientes, pero relacionados:

CU09: genera y registra la documentación contractual.

CU10: calcula y registra el cronograma de amortización.

Documentación de salida: reúne el acuerdo de condiciones, el comprobante de desembolso cuando proceda, la tabla de amortización y el contrato privado.

Una última precisión: los modelos de contrato deben ser revisados por un profesional jurídico familiarizado con la normativa boliviana y los requisitos regulatorios aplicables antes de utilizarlos como documentos contractuales reales.
